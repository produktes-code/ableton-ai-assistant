    def _get_key(params, ctx):
        song = ctx["song"]
        try:
            # API de Ableton 12
            return {
                "root_note": song.root_note,
                "scale_name": song.scale_name,
                "scale_mode": getattr(song, "scale_mode", None)
            }
        except AttributeError:
            # Ableton 11: no tiene esta API
            return {"root_note": None, "scale_name": None,
                    "note": "Requiere Ableton Live 12"}
    
    registry.register("set_key", _set_key, {
        "description": "Establece la tonalidad (Ableton 12)",
        "params": {
            "root_note": "int (0=C, 1=C#, ..., 11=B)",
            "scale_name": "str (ej: 'Major', 'Minor')"
        }
    })
    
    def _set_key(params, ctx):
        song = ctx["song"]
        try:
            if "root_note" in params:
                note = int(params["root_note"])
                song.root_note = max(0, min(11, note))
            if "scale_name" in params:
                song.scale_name = str(params["scale_name"])
            return {
                "root_note": song.root_note,
                "scale_name": song.scale_name
            }
        except AttributeError:
            return {"error": "Requiere Ableton Live 12"}
